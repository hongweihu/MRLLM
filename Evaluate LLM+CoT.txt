def evaluate_reward_function_with_llm(power_loss, heat_loss, prev_power_loss, prev_heat_loss,
                                      total_cost, h2_cost, 
                                      component_operation_cost, pv_curtailment_cost,
                                      client, model_name, session_id):
    """
    Dynamically evaluate performance based on improvement trends (vs previous iteration) 
    and domain knowledge regarding cost-loss trade-offs.
    """
    api_start = time.perf_counter()
    def calc_change_str(current, prev):
        if prev is None:
            return "N/A (First Iteration)"
        if prev == 0:
            return "N/A (Previous was 0)"
        change = ((current - prev) / prev) * 100
        sign = "+" if change > 0 else ""
        return f"{sign}{change:.2f}%"

    power_change_str = calc_change_str(power_loss, prev_power_loss)
    heat_change_str = calc_change_str(heat_loss, prev_heat_loss)
    
    prev_p_val = f"{prev_power_loss:.2f}" if prev_power_loss is not None else "None"
    prev_h_val = f"{prev_heat_loss:.2f}" if prev_heat_loss is not None else "None"

    evaluation_prompt = f"""
You are an expert in energy system optimization (Hydrogen-based Multi-energy System).
Please evaluate the Power Loss and Heat Loss reliability based on the current simulation results (30-day cumulative) compared to the previous iteration.

### Data Overview
1. Economic Indicators (30 days):
Total  Cost: {total_cost:.2f} Yuan
Hydrogen Cost: {h2_cost:.2f} Yuan
Equipment Operation Cost: {component_operation_cost:.2f} Yuan
PV Curtailment Cost: {pv_curtailment_cost:.2f} Yuan

2. Reliability Indicators (Target: Minimize Loss):
Power Loss: {power_loss:.2f} kWh (Previous: {prev_p_val} kWh, Change: {power_change_str})
Heat Loss: {heat_loss:.2f} kWh (Previous: {prev_h_val} kWh, Change: {heat_change_str})

### Evaluation Criteria
Please assess the "power_loss_evaluation" and "heat_loss_evaluation" using the labels: ["very good", "good", "not good", "bad"].

Judgment Logic:
1. Trend Analysis: Significant reduction (negative change %) is highly positive. 
   Increasing loss (positive change %) is negative.
2. Absolute Magnitude & Domain Knowledge: Compare the loss energy (kWh) with the Total Cost. 
c  Consider the trade-off: Is the system sacrificing a tiny bit of load reliability to save huge costs?

### Output Format
Reply ONLY in the following JSON format:
{{
    "power_loss_evaluation": "very good/good/not good/bad",
    "heat_loss_evaluation": "very good/good/not good/bad", 
    "reasoning": "Brief analysis of the trend (e.g., 'Reduced by 20%') and the magnitude relative to cost."
}}
"""
    
    messages = [
        {"role": "system", "content": "You are an energy system optimization expert."},
        {"role": "user", "content": evaluation_prompt}
    ]
    
    try:
        response = client.chat.completions.create(
            messages=messages,
            model=model_name,
            temperature=0.3,
            timeout=60
        )
        
        result = response.choices[0].message.content
        
        import re
        json_match = re.search(r'\{.*\}', result, re.DOTALL)
        if json_match:
            evaluation_result = json.loads(json_match.group(0))
            
            eval_save_path = Path("evaluation_results") / session_id
            eval_save_path.mkdir(parents=True, exist_ok=True)
            
            eval_file = eval_save_path / f"llm_evaluation_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
            with open(eval_file, 'w', encoding='utf-8') as f:
                json.dump(evaluation_result, f, indent=2, ensure_ascii=False)
            
            print(f"✅ LLM evaluation completed: P-Loss {power_change_str}, H-Loss {heat_change_str}")
            print(f"   Reasoning: {evaluation_result.get('reasoning', 'No reasoning')}")
            api_duration = time.perf_counter() - api_start
            return evaluation_result["power_loss_evaluation"], evaluation_result["heat_loss_evaluation"], api_duration
        else:
            print("❌ Failed to parse LLM evaluation results")
            return "not good", "not good", 0 
            
    except Exception as e:
        print(f"❌ LLM evaluation failed: {e}")
        return "not good", "not good", 0



# CoT suggestions for penalty function optimization
def generate_cot_suggestions(total_cost, h2_cost, component_operation_cost, pv_curtailment_cost, 
                             power_loss, heat_loss, power_loss_eval, heat_loss_eval,
                             best_code=None, best_loss_sum=None, current_loss_sum=None,
                             cost_eval=None, previous_total_cost=None,
                             relative_cost_change=None):
    """
    Generate Chain of Thought suggestions for rewriting the Python penalty function.
    """
    suggestions = []
    
    suggestions.append("Redesign the penalty function based on the given feedback. Let's think step by step:")
    suggestions.append("1. ANALYSIS: Check current performance against the zero-loss feasibility rule (power_loss <= 0 and heat_loss <= 0).")
    
    suggestions.append(f"   - Loss Status: Power_loss={power_loss:.2f} ({power_loss_eval}), Heat_loss={heat_loss:.2f} ({heat_loss_eval})")
    suggestions.append(f"   - Cost Breakdown: H2={h2_cost:.2f}, Ops={component_operation_cost:.2f}, PV_curt={pv_curtailment_cost:.2f}")
    if cost_eval is not None:
        suggestions.append(f"   - Cost Evaluation: {cost_eval}. This feedback is separate from the loss labels.")
    
    AGGRESSIVE = "Use a finite linear-plus-mild-power curve (power <= 1.5) and keep the multiplier between 5 and 40; never use exponential/cubic growth."
    MODERATE = "INCREASE the multiplier or slightly adjust the curve shape."
    SLIGHT_DEC = "SLIGHTLY DECREASE the penalty severity to prioritize cost savings."

    suggestions.append("2. ECONOMIC TRADE-OFF & RL OPTIMIZATION ANALYSIS:")
    if cost_eval == "increasing" and power_loss_eval == "very good" and heat_loss_eval == "very good":
        suggestions.append("   - COST-AWARE HISTORICAL FEEDBACK: reliability is already perfect but cost increased relative to the preceding round; reduce unnecessary conservatism while preserving non-positive loss.")
    # Detect an unstable low-cost/high-loss policy before the generic loss
    # branch, otherwise the old logic incorrectly recommended larger penalties.
    if total_cost < 100000 and (power_loss > 10000 or heat_loss > 10000):
        suggestions.append("   - DIAGNOSIS: The policy is exploiting an unstable reward trade-off (very low cost with massive loss). Keep a dense positive penalty for every positive per-step loss, but do not increase beyond a bounded 5-40 multiplier.")
        suggestions.append("   - STRATEGY: Use a linear term plus a mild power <= 1.5; avoid thresholds that make small per-step losses free.")
    # Check if penalties are too weak
    elif power_loss > 1000 or heat_loss > 1000:
        suggestions.append("   - DIAGNOSIS: The losses are massive (>1000). Your previous penalty was TOO WEAK compared to the cost of buying hydrogen/operating equipment.")
        suggestions.append("   - STRATEGY: Increase only within the bounded 5-40 range and keep the penalty active for every positive per-step loss. Do not use 100+ multipliers.")

    suggestions.append("3. IMBALANCE DETECTION:")
    # Check if it's over-optimizing one at the expense of the other
    if power_loss_eval == "very good" and heat_loss_eval == "bad":
        suggestions.append("   - DIAGNOSIS: IMBALANCE. The agent is perfectly fixing Power Loss but ignoring Heat Loss.")
        suggestions.append("   - STRATEGY: Increase the multiplier for the Heat Loss penalty to force the agent to care about it.")
    elif heat_loss_eval == "very good" and power_loss_eval == "bad":
        suggestions.append("   - DIAGNOSIS: IMBALANCE. The agent is perfectly fixing Heat Loss but ignoring Power Loss.")
        suggestions.append("   - STRATEGY: Increase the multiplier for the Power Loss penalty.")
    else:
        suggestions.append("   - No severe imbalance detected, or both are moving in the same direction.")

    suggestions.append("4. POWER PARAMETER STRATEGY:")
    if power_loss_eval == "bad":
        suggestions.append(f"   - CRITICAL: Power loss ({power_loss:.2f}) is SEVERELY above target (10). ACTION: {AGGRESSIVE}")
    elif power_loss_eval == "not good":
        suggestions.append(f"   - STATUS: Power loss ({power_loss:.2f}) is unacceptable but improving. ACTION: {MODERATE}")
    elif power_loss_eval == "good":
        if total_cost > 1e6:
            suggestions.append(f"   - STATUS: Power loss is controlled. ACTION: {SLIGHT_DEC} to improve economy.")
    else:
        suggestions.append("   - STATUS: Power loss is perfect. Focus entirely on optimizing heat parameters or reducing cost.")

    suggestions.append("5. HEAT PARAMETER STRATEGY:")
    if heat_loss_eval == "bad":
        suggestions.append(f"   - CRITICAL: Heat loss ({heat_loss:.2f}) is SEVERELY above target (10). ACTION: {AGGRESSIVE}")
    elif heat_loss_eval == "not good":
        suggestions.append(f"   - STATUS: Heat loss is not ideal. ACTION: {MODERATE}")
    elif heat_loss_eval == "good":
        if total_cost > 1e6:
            suggestions.append(f"   - STATUS: Heat loss controlled. ACTION: {SLIGHT_DEC} to improve economy.")
    else:
        suggestions.append("   - STATUS: Heat loss is well-controlled. Minimize heat penalty changes.")

    suggestions.append("6. MATH & STRUCTURAL RULES (CRITICAL FOR PPO):")
    suggestions.append("   - DO NOT use `math.exp`, `**3`, or `**4`. They are too unstable and cause gradient explosions.")
    suggestions.append("   - DO NOT use `min()` to cap the penalty. It creates zero gradients (flat lines) where the agent cannot learn.")
    suggestions.append("   - AVOID using boring piecewise linear functions (`if/elif/else`) that jump abruptly.")
    suggestions.append("   - Stick exclusively to smooth curves: `excess ** 1.2`, `excess ** 1.5`, or Huber-like combinations (quadratic near zero, linear for large values).")

    if power_loss_eval == "bad" and heat_loss_eval == "bad":
        suggestions.append("7. EMERGENCY PROTOCOL:")
        suggestions.append("   - SITUATION: Both Power and Heat constraints are failing completely.")
        suggestions.append("   - OVERRIDE: Do NOT worry about Total Cost. Drastically increase multipliers in BOTH functions using safe smooth curves.")
    suggestions.append("7. FINE-TUNING & BEST CODE PRESERVATION:")
    if best_loss_sum is not None and best_code is not None:
        if current_loss_sum <= best_loss_sum:
            suggestions.append("   - EXCELLENT: Your current code achieved a new BEST record (lowest combined losses)!")
            suggestions.append("   - STRATEGY: Make only VERY SMALL TWEAKS (e.g., adjust a multiplier by 5% or 10%). Do not completely rewrite the mathematical structure. We are almost there.")
        else:
            suggestions.append(f"   - WARNING: Your current code is WORSE than a previous version we found (Best Loss was {best_loss_sum:.2f}, but current is {current_loss_sum:.2f}).")
            suggestions.append("   - STRATEGY: Discard the current bad structure. Go back to a variation of the successful code. Here is the BEST code structure we found so far as reference:")
            suggestions.append(f"```python\n{best_code}\n```")
            suggestions.append("   - Please rewrite the penalty function based on this BEST code structure, but try slightly different multipliers.")
    if power_loss_eval == "very good" and heat_loss_eval == "very good":
        suggestions.append("7. ECONOMIC REFINEMENT:")
        if previous_total_cost is None or relative_cost_change is None:
            suggestions.append("   - This is the baseline round; retain the current cost as the historical reference.")
        elif relative_cost_change <= 0.005:
            suggestions.append("   - COST STABLE: the absolute change from the immediately preceding round is <= 0.5%; make only a bounded local change or stop if all safety gates pass.")
        elif total_cost < previous_total_cost:
            suggestions.append("   - COST IMPROVING: the measured cost decreased relative to the immediately preceding round; preserve zero loss and continue conservative local refinement.")
        else:
            suggestions.append("   - COST INCREASING: the measured cost increased relative to the immediately preceding round; reduce unnecessary penalty severity without permitting positive loss.")
                
            
    suggestions.extend([
        "8. FINAL GUIDELINES:",
        "- Output ONLY the Python code. No explanations or comments outside the code block.",
        "- Do NOT change the function names.",
        "- Feasibility (power_loss <= 0 and heat_loss <= 0) > Economy. First obtain non-positive loss, then use only adjacent-round cost changes for refinement."
    ])
    
    return suggestions
